# Bootstrap de grupos/permissões — LuckPerms
#
# Execute estes comandos EM ORDEM no console do servidor (ou via RCON),
# depois que o LuckPerms já tiver bootado pelo menos uma vez.
# Cada linha é um comando "/lp ...". Não copie os comentários (#).
#
# Contexto/decisões: ver ../../../docs/fundacao-permissoes.md

# ============================================================
# Nível 0 — Aventureiro (base permanente de todo reencarnado)
# ============================================================
lp group default permission set pangeia.tier.0 true
lp group default permission set pangeia.profissao.aventureiro true

# Política de locomoção: sem /spawn, /home, /warp, /tpa, /back por padrão
lp group default permission set essentials.spawn false
lp group default permission set essentials.home false
lp group default permission set essentials.sethome false
lp group default permission set essentials.warp false
lp group default permission set essentials.tpa false
lp group default permission set essentials.tpahere false
lp group default permission set essentials.tpaccept false
lp group default permission set essentials.back false

# Staff — libera de volta os comandos de locomoção pra moderação/admin
# ============================================================

lp group staff parent add default
lp group staff permission set essentials.spawn true
lp group staff permission set essentials.home true
lp group staff permission set essentials.sethome true
lp group staff permission set essentials.warp true
lp group staff permission set essentials.tpa true
